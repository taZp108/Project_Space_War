data ProcessorArchitecture = PaUnknown WORD | PaIntel | PaMips | PaAlpha | PaPpc | PaIa64 | PaIa32OnIa64 | PaAmd64
    deriving (Show,Eq)

instance Storable ProcessorArchitecture where
    sizeOf _ = sizeOf (undefined::WORD)
    alignment _ = alignment (undefined::WORD)
    poke buf pa = pokeByteOff buf 0 $ case pa of
        PaUnknown w -> w
        PaIntel     -> 0
{-# LINE 138 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        PaMips      -> 1
{-# LINE 139 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        PaAlpha     -> 2
{-# LINE 140 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        PaPpc       -> 3
{-# LINE 141 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        PaIa64      -> 6
{-# LINE 142 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}

{-# LINE 143 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        PaIa32OnIa64 -> 10
{-# LINE 144 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}

{-# LINE 145 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        PaAmd64     -> 9
{-# LINE 146 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
    peek buf = do
        v <- (peekByteOff buf 0:: IO WORD)
        return $ case v of
            (0) -> PaIntel
{-# LINE 150 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
            (1)  -> PaMips
{-# LINE 151 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
            (2) -> PaAlpha
{-# LINE 152 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
            (3)   -> PaPpc
{-# LINE 153 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
            (6)  -> PaIa64
{-# LINE 154 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}

{-# LINE 155 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
            (10) -> PaIa32OnIa64
{-# LINE 156 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}

{-# LINE 157 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
            (9) -> PaAmd64

            w                                   -> PaUnknown w

data SYSTEM_INFO = SYSTEM_INFO
    { siProcessorArchitecture :: ProcessorArchitecture
    , siPageSize :: DWORD
    , siMinimumApplicationAddress, siMaximumApplicationAddress :: LPVOID
    , siActiveProcessorMask :: DWORD
    , siNumberOfProcessors :: DWORD
    , siProcessorType :: DWORD
    , siAllocationGranularity :: DWORD
    , siProcessorLevel :: WORD
    , siProcessorRevision :: WORD
    } deriving (Show)

instance Storable SYSTEM_INFO where
    sizeOf = const (48)
{-# LINE 174 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
    alignment _ = 8
{-# LINE 175 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
    poke buf si = do
        ((\hsc_ptr -> pokeByteOff hsc_ptr 0)) buf (siProcessorArchitecture si)
{-# LINE 177 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        ((\hsc_ptr -> pokeByteOff hsc_ptr 4))             buf (siPageSize si)
{-# LINE 178 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        ((\hsc_ptr -> pokeByteOff hsc_ptr 8)) buf (siMinimumApplicationAddress si)
{-# LINE 179 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        ((\hsc_ptr -> pokeByteOff hsc_ptr 16)) buf (siMaximumApplicationAddress si)
{-# LINE 180 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        ((\hsc_ptr -> pokeByteOff hsc_ptr 24))  buf (siActiveProcessorMask si)
{-# LINE 181 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        ((\hsc_ptr -> pokeByteOff hsc_ptr 32))   buf (siNumberOfProcessors si)
{-# LINE 182 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        ((\hsc_ptr -> pokeByteOff hsc_ptr 36))        buf (siProcessorType si)
{-# LINE 183 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        ((\hsc_ptr -> pokeByteOff hsc_ptr 40)) buf (siAllocationGranularity si)
{-# LINE 184 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        ((\hsc_ptr -> pokeByteOff hsc_ptr 44))        buf (siProcessorLevel si)
{-# LINE 185 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        ((\hsc_ptr -> pokeByteOff hsc_ptr 46))     buf (siProcessorRevision si)
{-# LINE 186 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}

    peek buf = do
        processorArchitecture <-
            ((\hsc_ptr -> peekByteOff hsc_ptr 0)) buf
{-# LINE 190 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        pageSize            <- ((\hsc_ptr -> peekByteOff hsc_ptr 4)) buf
{-# LINE 191 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        minimumApplicationAddress <-
            ((\hsc_ptr -> peekByteOff hsc_ptr 8)) buf
{-# LINE 193 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        maximumApplicationAddress <-
            ((\hsc_ptr -> peekByteOff hsc_ptr 16)) buf
{-# LINE 195 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        activeProcessorMask <- ((\hsc_ptr -> peekByteOff hsc_ptr 24)) buf
{-# LINE 196 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        numberOfProcessors  <- ((\hsc_ptr -> peekByteOff hsc_ptr 32)) buf
{-# LINE 197 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        processorType       <- ((\hsc_ptr -> peekByteOff hsc_ptr 36)) buf
{-# LINE 198 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        allocationGranularity <-
            ((\hsc_ptr -> peekByteOff hsc_ptr 40)) buf
{-# LINE 200 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        processorLevel      <- ((\hsc_ptr -> peekByteOff hsc_ptr 44)) buf
{-# LINE 201 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        processorRevision   <- ((\hsc_ptr -> peekByteOff hsc_ptr 46)) buf
{-# LINE 202 "libraries\\Win32\\System\\Win32\\Info\\Internal.hsc" #-}
        return $ SYSTEM_INFO {
            siProcessorArchitecture     = processorArchitecture,
            siPageSize                  = pageSize,
            siMinimumApplicationAddress = minimumApplicationAddress,
            siMaximumApplicationAddress = maximumApplicationAddress,
            siActiveProcessorMask       = activeProcessorMask,
            siNumberOfProcessors        = numberOfProcessors,
            siProcessorType             = processorType,
            siAllocationGranularity     = allocationGranularity,
            siProcessorLevel            = processorLevel,
            siProcessorRevision         = processorRevision
            }
